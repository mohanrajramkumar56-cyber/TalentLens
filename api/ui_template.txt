  <script>
    const $ = (id) => document.getElementById(id);
    const CONV_KEY = \"talentlens_conversations\";
    const ACTIVE_KEY = \"talentlens_active\";

    // ---------- conversation persistence ----------
    function loadConvs() { try { return JSON.parse(localStorage.getItem(CONV_KEY)) || []; } catch { return []; } }
    function saveConvs(c) { localStorage.setItem(CONV_KEY, JSON.stringify(c)); }
    function activeId() { return localStorage.getItem(ACTIVE_KEY); }
    function setActiveId(id) { localStorage.setItem(ACTIVE_KEY, id); }

    let conversations = loadConvs();

    function currentConv() {
      let id = activeId();
      if (!conversations.find(c => c.id === id)) {
        if (!conversations.length) id = newConv(false);
        else id = conversations[0].id;
      }
      setActiveId(id);
      return conversations.find(c => c.id === id) || newConv(true);
    }

    function newConv(toggle = true) {
      const id = Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
      conversations.unshift({ id, title: \"New conversation\", messages: [] });
      saveConvs(conversations);
      if (toggle) { setActiveId(id); renderSidebar(); renderMessages(); }
      return conversations.find(c => c.id === id);
    }

    function deleteConv(id) {
      conversations = conversations.filter(c => c.id !== id);
      saveConvs(conversations);
      if (activeId() === id) {
        if (conversations.length) setActiveId(conversations[0].id); else newConv(true);
        renderMessages();
      }
      renderSidebar();
    }

    function clearAll() {
      conversations = [];
      saveConvs(conversations);
      newConv(true);
      renderSidebar();
    }

    function appendMessage(role, content) {
      const c = currentConv();
      c.messages.push({ role, content });
      if (c.messages.length === 1 && role === \"user\") {
        c.title = content.slice(0, 32) + (content.length > 32 ? \"…\" : \"\");
      }
      saveConvs(conversations);
      renderSidebar();
    }

    // ---------- sidebar ----------
    function renderSidebar() {
      const list = $(\"convList\");
      list.innerHTML = \"\";
      if (!conversations.length) {
        list.innerHTML = '<div class=\"text-sm text-slate-500 text-center py-8\">No conversations yet</div>';
        return;
      }
      conversations.forEach(c => {
        const el = document.createElement(\"div\");
        const active = c.id === activeId();
        el.className = \"group flex items-center gap-2 px-3 py-2.5 rounded-xl cursor-pointer text-sm transition \" +
          (active ? \"bg-indigo-600/20 border border-indigo-500/30 text-white\" : \"text-slate-300 hover:bg-white/5\");
        el.innerHTML = \`<span class=\"flex-1 truncate\">📌 ${c.title}</span>\` +
          \`<button class=\"hidden group-hover:block text-slate-500 hover:text-red-400\" data-del=\"${c.id}\">✕</button>\`;
        el.onclick = () => { setActiveId(c.id); renderSidebar(); renderMessages(); };
        el.querySelector(\"[data-del]\").onclick = (e) => { e.stopPropagation(); deleteConv(c.id); };
        list.appendChild(el);
      });
    }

    // ---------- messages ----------
    function renderMessages() {
      const box = $(\"messages\");
      box.innerHTML = \"\";
      const c = currentConv();
      c.messages.forEach(m => paintMsg(m.role, m.content));
      box.scrollTop = box.scrollHeight;
    }

    function paintMsg(role, content, isErr = false) {
      const box = $(\"messages\");
      const row = document.createElement(\"div\");
      row.className = \"flex \" + (role === \"user\" ? \"justify-end\" : \"justify-start\") + \" gap-2 items-start\";

      if (role === \"assistant\") {
        const avatar = document.createElement(\"div\");
        avatar.className = \"w-8 h-8 rounded-xl bg-gradient-to-br from-indigo-500 to-purple-600 flex items-center justify-center text-sm font-extrabold text-white shrink-0 mt-0.5\";
        avatar.textContent = \"T\";
        row.appendChild(avatar);
      }

      const bubble = document.createElement(\"div\");
      bubble.className = isErr
        ? \"max-w-[80%] px-4 py-2.5 rounded-2xl rounded-tl-sm bg-red-500/15 border border-red-500/20 text-red-200 whitespace-pre-wrap\"
        : role === \"user\"
          ? \"max-w-[80%] px-4 py-2.5 rounded-2xl rounded-tr-sm bg-indigo-600 text-white whitespace-pre-wrap\"
          : \"max-w-[80%] px-4 py-2.5 rounded-2xl rounded-tl-sm bg-white/5 border border-white/10 text-slate-100 whitespace-pre-wrap\";
      bubble.textContent = content;
      row.appendChild(bubble);
      box.appendChild(row);
      box.scrollTop = box.scrollHeight;
      return bubble;
    }

    // ---------- status / stages ----------
    const setStatus = (text, cls) => {
      const s = $(\"status\");
      s.textContent = text;
      s.className = \"px-3 py-1 rounded-full text-xs font-semibold \" + cls;
    };

    document.addEventListener(\"DOMContentLoaded\", () => {
      renderSidebar();
      const c = currentConv();
      if (c.messages.length) {
        $(\"uploadStage\").classList.add(\"hidden\");
        $(\"chatStage\").classList.remove(\"hidden\");
        renderMessages();
        fetch(\"/api/status\").then(r => r.json()).then(d => {
          if (d.ready) setStatus(\"Ready — ask away\", \"bg-emerald-500/15 text-emerald-300 border border-emerald-500/20\");
        }).catch(() => {});
      }
    });

    // ---------- upload ----------
    const dropZone = $(\"dropZone\"), filesInput = $(\"files\"), fileCount = $(\"fileCount\");

    dropZone.addEventListener(\"click\", () => filesInput.click());
    filesInput.addEventListener(\"change\", () => {
      const n = filesInput.files.length;
      fileCount.textContent = n ? \`${n} file${n > 1 ? \"s\" : \"\"} selected — ready to process\` : \"Drag & drop resumes here\";
    });
    [\"dragover\", \"dragleave\", \"drop\"].forEach(ev => dropZone.addEventListener(ev, e => {
      e.preventDefault();
      if (ev !== \"dragover\") dropZone.classList.remove(\"border-indigo-400\", \"bg-white/5\");
      else dropZone.classList.add(\"border-indigo-400\", \"bg-white/5\");
    }));
    dropZone.addEventListener(\"drop\", e => {
      filesInput.files = e.dataTransfer.files;
      const n = filesInput.files.length;
      fileCount.textContent = n ? \`${n} file${n > 1 ? \"s\" : \"\"} selected — ready to process\` : \"Drag & drop resumes here\";
    });

    $(\"processBtn\").addEventListener(\"click\", async () => {
      if (!filesInput.files.length) { fileCount.textContent = \"⚠ Please choose at least one PDF first.\"; return; }
      const btn = $(\"processBtn\"); btn.disabled = true; btn.textContent = \"Processing…\";
      $(\"uploadMsg\").textContent = \"Uploading & building index… this can take a bit\";
      const fd = new FormData();
      for (const f of filesInput.files) fd.append(\"files\", f);
      try {
        const r = await fetch(\"/api/upload\", { method: \"POST\", body: fd });
        const data = await r.json();
        if (!r.ok) throw new Error(data.error || \"Upload failed.\");
        $(\"uploadMsg\").textContent = \"\";
        fileCount.textContent = \`✓ ${data.files} file(s) indexed (${data.chunks} chunks)\`;
        setStatus(\"Ready — ask away\", \"bg-emerald-500/15 text-emerald-300 border border-emerald-500/20\");
        $(\"uploadStage\").classList.add(\"hidden\");
        $(\"chatStage\").classList.remove(\"hidden\");
        $(\"q\").focus();
      } catch (e) {
        $(\"uploadMsg\").textContent = e.message;
      } finally {
        btn.disabled = false; btn.textContent = \"⚡ Process Resumes\";
      }
    });

    // ---------- chat ----------
    $(\"chatForm\").addEventListener(\"submit\", async (e) => {
      e.preventDefault();
      const q = $(\"q\").value.trim();
      if (!q) return;
      paintMsg(\"user\", q);
      appendMessage(\"user\", q);
      $(\"q\").value = \"\";
      $(\"sendBtn\").disabled = true;

      const typing = paintMsg(\"assistant\", \"Thinking\");
      const dots = document.createElement(\"span\");
      dots.innerHTML = '<span class=\"dot\">.</span><span class=\"dot\">.</span><span class=\"dot\">.</span>';
      typing.appendChild(dots);

      try {
        const r = await fetch(\"/api/chat\", {
          method: \"POST\",
          headers: { \"Content-Type\": \"application/json\" },
          body: JSON.stringify({ question: q }),
        });
        const data = await r.json();
        if (!r.ok) throw new Error(data.error || \"Chat failed.\");
        typing.textContent = data.answer;
        appendMessage(\"assistant\", data.answer);
      } catch (err) {
        typing.textContent = err.message;
        typing.className = \"max-w-[80%] px-4 py-2.5 rounded-2xl bg-red-500/15 border border-red-500/20 text-red-200 whitespace-pre-wrap\";
      } finally {
        $(\"sendBtn\").disabled = false;
      }
    });

    // ---------- navbar actions ----------
    $(\"newChatBtn\").addEventListener(\"click\", async () => {
      newConv(true);
      $(\"uploadStage\").classList.remove(\"hidden\");
      $(\"chatStage\").classList.add(\"hidden\");
      filesInput.value = \"\";
      fileCount.textContent = \"Drag & drop resumes here\";
      setStatus(\"No resumes loaded\", \"bg-white/5 border border-white/10 text-slate-300\");
      try { await fetch(\"/api/reset\", { method: \"POST\" }); } catch (_) {}
    });
    $(\"clearAllBtn\").addEventListener(\"click\", clearAll);
    $(\"sidebarToggle\").addEventListener(\"click\", () => $(\"sidebar\").classList.toggle(\"hidden\"));

    // ---------- 3D tilt ----------
    const card = $(\"mainCard\");
    card.addEventListener(\"mousemove\", e => {
      const r = card.getBoundingClientRect();
      const x = (e.clientX - r.left) / r.width - 0.5;
      const y = (e.clientY - r.top) / r.height - 0.5;
      card.style.transform = \`perspective(1200px) rotateY(${x * 4}deg) rotateX(${-y * 4}deg)\`;
    });
    card.addEventListener(\"mouseleave\", () => { card.style.transform = \"perspective(1200px) rotateY(0) rotateX(0)\"; });

    // ---------- particle network ----------
    const canvas = $(\"particles\"), ctx = canvas.getContext(\"2d\");
    let W, H, pts;
    function resize() { W = canvas.width = innerWidth; H = canvas.height = innerHeight; }
    resize(); addEventListener(\"resize\", resize);
    pts = Array.from({ length: 60 }, () => ({ x: Math.random() * W, y: Math.random() * H, vx: (Math.random() - .5) * .5, vy: (Math.random() - .5) * .5 }));
    setInterval(() => { if (pts.length < 60) { pts = Array.from({ length: 60 }, () => ({ x: Math.random() * W, y: Math.random() * H, vx: (Math.random()-.5)*.5, vy: (Math.random()-.5)*.5 })); } }, 1000);
    (function loop() {
      ctx.clearRect(0, 0, W, H);
      for (const p of pts) {
        p.x += p.vx; p.y += p.vy;
        if (p.x < 0 || p.x > W) p.vx *= -1;
        if (p.y < 0 || p.y > H) p.vy *= -1;
      }
      for (let i = 0; i < pts.length; i++) {
        for (let j = i + 1; j < pts.length; j++) {
          const d = Math.hypot(pts[i].x - pts[j].x, pts[i].y - pts[j].y);
          if (d < 120) { ctx.strokeStyle = \`rgba(129,140,248,${.12 * (1 - d / 120)})\`; ctx.lineWidth = 1; ctx.beginPath(); ctx.moveTo(pts[i].x, pts[i].y); ctx.lineTo(pts[j].x, pts[j].y); ctx.stroke(); }
        }
      }
      for (const p of pts) { ctx.fillStyle = \"rgba(196,181,253,.6)\"; ctx.beginPath(); ctx.arc(p.x, p.y, 1.6, 0, Math.PI * 2); ctx.fill(); }
      requestAnimationFrame(loop);
    })();
  </script>
